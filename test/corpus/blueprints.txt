==================
Blueprint Definition with Attributes and Methods
==================
blueprint Point {
    int x;
    int y;

    forge sum() ~> int {
        return my.x + my.y;
    }

    forge scale(int k) {
        my.x = my.x * k;
        my.y = my.y * k;
    }

    forge describe() {
        print("Point");
    }
}
